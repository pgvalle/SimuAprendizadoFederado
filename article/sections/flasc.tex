\section{FLASC} \label{sec:proposta}

O algoritmo FLASC (\textit{Federated Learning with Adaptive Soft Clustering}) estende o FLSC \cite{li2022federated}
ao introduzir um mecanismo de atribuição de cluster mais flexível e sensível à distribuição local.
O processo inicia-se no servidor, que mantém um conjunto de $K$ modelos globais $\mathcal{W} = \{W_{1}, W_{2}, \dots, W_{K}\}$.
Em cada rodada de treinamento $t$, o servidor seleciona aleatoriamente uma fração de clientes participantes e transmite os $K$ modelos a eles.

Ao receber os modelos, o cliente $c$ avalia a perda empírica de cada um utilizando seu conjunto de dados local $\mathcal{D}_c$.
A Equação \ref{eq:loss_eval} descreve a obtenção da perda $L_{c,k}$ do $k$-ésimo modelo no cliente $c$.
\begin{equation}
    L_{c,k} = \text{Loss}(W_{k}; \mathcal{D}_c) \label{eq:loss_eval}
\end{equation}

A etapa de seleção depende da dispersão das perdas observadas, onde o cliente define $L_{min} = \min_k \{L_{c,k}\}$ e $L_{max} = \max_k \{L_{c,k}\}$.
Caso a diferença entre o melhor e o pior desempenho seja desprezível ($L_{max} - L_{min} \leq \epsilon$), o algoritmo trata os modelos como indistinguíveis.
Caso contrário, a Equação \ref{eq:normalization} normaliza as perdas no intervalo $[0, 1]$.
\begin{equation}
    \hat{L}_{c,k} = \frac{L_{c,k} - L_{min}}{L_{max} - L_{min}} \label{eq:normalization}
\end{equation}

Dessa forma, o cliente define o conjunto de índices $S_c$ dos modelos selecionados de acordo com
a sensibilidade $\epsilon$ e o fator de tolerância $\rho$ (Equação \ref{eq:selection_set}).
Em casos de indistinguibilidade, o cliente seleciona aleatoriamente um único modelo.
\begin{equation}
    S_c = \begin{cases}
      \{k'\}, \text{ com } k' \sim \text{Unif}\{1, \dots, K\}, & \text{se } L_{max} - L_{min} \leq \epsilon \\
      \{k \mid \hat{L}_{c,k} \leq \rho\}, & \text{caso contrário}
    \end{cases} \label{eq:selection_set}
\end{equation}

Com os modelos selecionados, o FLASC atribui pesos $\alpha_{c,k}$ que decrescem linearmente com as perdas normalizadas (Equação \ref{eq:fusion_weights}).
Note que, se o cliente seleciona um único modelo, este deve receber peso total ($1$).
\begin{equation}
    \alpha_{c,k} = \frac{1 - L_{c,k}}{\sum_{j \in S_c}(1 - L_{c,j})} \label{eq:fusion_weights}
\end{equation}

A Listagem \ref{code:flasc_select} apresenta a implementação em Python da lógica de seleção.
O código recebe o vetor das perdas de cada modelo, $\rho$ e $\epsilon$, e retorna o conjunto $S_c$ e os pesos de fusão.

\begin{lstlisting}[language=Python, caption={Lógica de seleção adaptativa no cliente FLASC.},
label={code:flasc_select}, frame=lines, numbers=left, basicstyle=\footnotesize\ttfamily, keywordstyle=\color{blue}]
def select_models(losses, rho, epsilon=1e-5):
    lmin, lmax = min(losses), max(losses)

    # Caso de indistinguibilidade
    if (lmax - lmin) <= epsilon:
        k = np.random.choice(len(losses))
        return [k], [1.0]

    # Selecao baseada em tolerancia (rho)
    Sc, raw_weights = [], []
    for k, lck in enumerate(losses):
        # Normalizacao das perdas
        norm_lck = (lck - lmin) / (lmax - lmin)
        if norm_lck <= rho:
            Sc.append(k)
            # melhor perda = maior peso
            raw_weights.append(1.0 - norm_lck)

    # calculo dos pesos
    total = sum(raw_weights)
    weights = [w / total for w in raw_weights]
    return Sc, weights
\end{lstlisting}

Após a seleção, o cliente constrói o modelo de partida para o treinamento local $W_{c}^{start}$
através da agregação dos modelos selecionados ($W_{c}^{start} = \sum_{k \in S_c} \alpha_{c,k} W_{k}$).
Após o treinamento por $E$ épocas, o cliente envia o modelo resultante $W_{c}^{end}$ e o conjunto de identificadores $S_c$ ao servidor.

O servidor atualiza os modelos globais de forma independente.
Para cada cluster $k$, o servidor agrega apenas as atualizações dos clientes que incluíram o modelo $k$ em sua fusão ($k \in S_c$),
seguindo a Equação \ref{eq:aggregation}, onde $n_c$ representa o número de amostras locais.
\begin{equation}
    W_{k}^{new} = \frac{\sum_{c: k \in S_c} n_c W_{c}^{end}}{\sum_{c: k \in S_c} n_c} \label{eq:aggregation}
\end{equation}

Esta arquitetura permite que os clusters evoluam para representar especializações distintas, enquanto a natureza \textit{soft} da seleção garante que os clientes se beneficiem de múltiplos domínios de conhecimento.
O hiperparâmetro $\rho$ regula essa generalização: valores próximos a 0 aproximam o FLASC de abordagens de \textit{hard clustering} como o IFCA \cite{ghosh2020efficient}, enquanto valores próximos a 1 replicam a fusão inclusiva do FLSC \cite{li2022federated}.
O Algoritmo \ref{alg:flasc} detalha o fluxo operacional completo.

\begin{algorithm}[H]
    \SetAlgoLined
    \KwIn{Parâmetros iniciais $w_0$, tolerância $\rho$, sensibilidade $\epsilon$}
    \KwData{Conjuntos de dados locais $\{\mathcal{D}_c\}$ nos clientes}
    \BlankLine
    \textbf{Servidor:} Inicializa $K$ modelos de cluster $\mathcal{W}_0 = \{W_{1,0}, \dots, W_{K,0}\}$ com $w_0$\;
    \For{cada rodada $t = 0, 1, \dots, T$}{
        Servidor seleciona aleatoriamente um subconjunto de clientes $P_t$\;
        Servidor transmite $\mathcal{W}_t$ para todos os clientes em $P_t$\;
        \BlankLine
        \textbf{Cada cliente $c \in P_t$ em paralelo:}\;
        Avalia perdas locais $L_{c,k} = \text{Loss}(W_{k,t}; \mathcal{D}_c)$ para $k \in \{1, \dots, K\}$\;
        $L_{min} \leftarrow \min_k \{L_{c,k}\}$, $L_{max} \leftarrow \max_k \{L_{c,k}\}$\;
        \If{$L_{max} - L_{min} \leq \epsilon$}{
            $S_c \leftarrow \{ \text{índice aleatório de } \{1, \dots, K\} \}$\;
            $\alpha_{c,k} \leftarrow 1.0$ para $k \in S_c$\;
        }
        \Else{
            $\hat{L}_{c,k} \leftarrow \frac{L_{c,k} - L_{min}}{L_{max} - L_{min}}$ para cada $k$\;
            $S_c \leftarrow \{k \mid \hat{L}_{c,k} \leq \rho\}$\;
            $v_{c,k} \leftarrow 1 - \hat{L}_{c,k}$ para $k \in S_c$\;
            $\alpha_{c,k} \leftarrow v_{c,k} / \sum_{j \in S_c} v_{c,j}$ para $k \in S_c$\;
        }
        $W_{c,t}^{start} \leftarrow \sum_{k \in S_c} \alpha_{c,k} W_{k,t}$\;
        $W_{c,t}^{end} \leftarrow \text{TreinamentoLocal}(W_{c,t}^{start}, \mathcal{D}_c)$\;
        Envia $\{W_{c,t}^{end}, S_c\}$ para o servidor\;
        \BlankLine
        \textbf{Servidor:}\;
        \For{cada cluster $k \in \{1, \dots, K\}$}{
            $W_{k,t+1} \leftarrow \frac{\sum_{c: k \in S_c} n_c W_{c,t}^{end}}{\sum_{c: k \in S_c} n_c}$\;
        }
    }
    \caption{Algoritmo FLASC}
    \label{alg:flasc}
\end{algorithm}
